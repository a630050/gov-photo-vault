快速串接範例

# 1. 查詢目前雙層身分能力與上限
curl -H "Authorization: Bearer <ACCESS_TOKEN>" \
  {{ base_url }}/integration/v1/capabilities

# 2. 搜尋授權資料集照片（預設時區 Asia/Taipei，迄日含當天）
curl -X POST {{ base_url }}/integration/v1/photos/search \
  -H "Authorization: Bearer <ACCESS_TOKEN>" \
  -H "Content-Type: application/json" \
  -d '{"mode":"general","from":"2026-06-01","to":"2026-06-30","limit":12}'

# 3. 呼叫 HTTP MCP 列出可用唯讀工具
curl -X POST {{ base_url }}/mcp \
  -H "Authorization: Bearer <ACCESS_TOKEN>" \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":1,"method":"tools/list"}'
